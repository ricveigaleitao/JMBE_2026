# Compute Wilcoxon p per measure (two levels in Culture)

wilcox_res <- do.call(rbind, lapply(split(alpha_df, alpha_df$variable), function(df) {
  df <- df[complete.cases(df$value, df$Culture), ]
  wt <- wilcox.test(value ~ Culture, data = df, exact = FALSE, correct = TRUE)
  data.frame(
    variable    = unique(df$variable),
    p           = wt$p.value,
    group1      = levs[1],
    group2      = levs[2],
    stringsAsFactors = FALSE
  )
}))
